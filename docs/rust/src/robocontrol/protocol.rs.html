1//! Protocolo serial do braço 4-DOF (firmware v1.3+).
2//!
3//! Enlace USB CDC a **115200 baud, 8N1**, linhas terminadas em `\n`.
4//! O PC nunca gera PWM: ele só propõe ALVOS. A rampa, os limites mecânicos e o
5//! hold continuam no firmware, então um comando perdido não vira tranco.
6//!
7//! # Convenções
8//!
9//! - Juntas sempre na ordem **Base, Ombro, Cotovelo, Garra**, em graus inteiros.
10//! - Sticks virtuais em `-100..=100` por eixo; o eixo Y chega invertido em
11//!   relação à tela (ver [`cmd_sticks`]).
12//! - Comandos começam com `#`; mensagens da placa começam com `S` ou `B`.
13//!
14//! # Comandos (PC → placa)
15//!
16//! | Item | Bytes | Efeito |
17//! |---|---|---|
18//! | [`CMD_ASSUME`] | `#M\n` | adota o controle do PC |
19//! | [`CMD_RELEASE`] | `#J\n` | devolve o controle ao joystick |
20//! | [`CMD_STOP`] | `#H\n` | segura as juntas onde estão |
21//! | [`CMD_DIAG`] | `#B\n` | pede o diagnóstico |
22//! | [`cmd_lock`] | `#L1\n` / `#L0\n` | trava (só aperta) / libera a garra |
23//! | [`cmd_targets`] | `#P...` | alvos absolutos por junta |
24//! | [`cmd_sticks`] | `#V...` | sticks virtuais |
25//!
26//! # Mensagens (placa → PC)
27//!
28//! | Prefixo | Tipo | Formato |
29//! |---|---|---|
30//! | `S` | [`Status`] | `S<pose0>;<pose1>;<pose2>;<pose3>;<pc_mode>;<trava>` |
31//! | `B` | [`Diag`] | `B<versão>;boots=<n>;limb=<min>;<max>;limo=...;limc=...;limg=...` |
32//! | outro | [`Msg::Text`] | banner, ajuda ou debug |
33//!
34//! O campo `trava` do status é opcional: firmware anterior não o enviava e o
35//! parser assume `false`.
36
37pub const CMD_ASSUME: &str = "#M\n";
38pub const CMD_RELEASE: &str = "#J\n";
39pub const CMD_STOP: &str = "#H\n";
40
41/// Trava de pressão da garra: com ela ativa a garra só aperta, nunca solta.
42///
43/// É o mesmo estado do botão físico D4 no firmware, então app e bancada nunca
44/// discordam. Enquanto travada, nem `#P` nem `#V` conseguem abrir a garra.
45pub fn cmd_lock(travar: bool) -> &'static str {
46    if travar { "#L1\n" } else { "#L0\n" }
47}
48
49/// Pedido de diagnóstico. A placa responde com [`Diag`] em [`Msg::Diag`].
50pub const CMD_DIAG: &str = "#B\n";
51
52/// `#P90;60;45;20\n` — alvos absolutos em graus, na ordem
53/// Base, Ombro, Cotovelo, Garra.
54///
55/// O firmware interpola até esses alvos com a própria rampa; valores são
56/// arredondados aqui porque o protocolo transporta inteiros.
57pub fn cmd_targets(pose: [f32; 4]) -> String {
58    format!(
59        "#P{};{};{};{}\n",
60        pose[0].round() as i32,
61        pose[1].round() as i32,
62        pose[2].round() as i32,
63        pose[3].round() as i32
64    )
65}
66
67/// `#V<lx>;<ly>;<rx>;<ry>\n` — sticks virtuais, `-100..=100` por eixo.
68///
69/// Entram no firmware pelo mesmo caminho dos joysticks físicos, então respeitam
70/// a mesma rampa e os mesmos limites. Lx/Ly são o pad esquerdo
71/// (Base/Cotovelo) e Rx/Ry o direito (Garra/Ombro); Y positivo é *para cima*,
72/// por isso o chamador nega o Y da tela. Valores fora da faixa são grampeados.
73pub fn cmd_sticks(v: [i32; 4]) -> String {
74    format!("#V{};{};{};{}\n", v[0].clamp(-100, 100), v[1].clamp(-100, 100), v[2].clamp(-100, 100), v[3].clamp(-100, 100))
75}

77/// Estado periódico enviado pelo firmware (~20 Hz).
78#[derive(Debug, Clone, PartialEq)]
79pub struct Status {
80    /// Pose medida, em graus: Base, Ombro, Cotovelo, Garra.
81    pub pose: [f32; 4],
82    /// `true` quando o firmware está aceitando alvos do PC.
83    pub pc_mode: bool,
84    /// `true` quando a garra está travada com pressão (botão D4 ou app).
85    pub trava: bool,
86}

88/// Resposta de [`CMD_DIAG`]: identidade, contador de boots e limites mecânicos.
89///
90/// `boots` só cresce enquanto a placa está ligada; um salto inesperado entre
91/// duas leituras indica reinício (o app usa isso para avisar sobre alimentação).
92#[derive(Debug, Clone, PartialEq)]
93pub struct Diag {
94    pub version: String,
95    pub boots: u32,
96    /// `[[min, max]; 4]` na ordem Base, Ombro, Cotovelo, Garra.
97    pub limits: [[f32; 2]; 4],
98}
99
100/// Uma linha já interpretada da serial.
101#[derive(Debug, Clone, PartialEq)]
102pub enum Msg {
103    Status(Status),
104    Diag(Diag),
105    /// Texto livre do firmware (banner, ajuda, debug).
106    Text(String),
107}
108
109/// Converte uma linha recebida em mensagem. Nunca entra em pânico com lixo.
110///
111/// Linha vazia (ou só `\r\n`) vira `None`; prefixo `S`/`B` malformado também é
112/// descartado; qualquer outra coisa vira [`Msg::Text`] para aparecer no log.
113pub fn parse_line(line: &str) -> Option<Msg> {
114    let line = line.trim_end_matches(['\r', '\n']).trim();
115    if line.is_empty() {
116        return None;
117    }
118    match line.as_bytes()[0] {
119        b'S' => parse_status(line).map(Msg::Status),
120        b'B' => parse_diag(line).map(Msg::Diag),
121        _ => Some(Msg::Text(line.to_string())),
122    }
123}
124
125/// `S<pose0>;...;<pose3>;<pc_mode>[;<trava>]`.
126///
127/// Falha (`None`) se faltar qualquer um dos cinco campos obrigatórios ou se um
128/// número não parsear; campos extras são ignorados.
129fn parse_status(line: &str) -> Option<Status> {
130    let mut it = line[1..].split(';');
131    let mut pose = [0.0; 4];
132    for p in pose.iter_mut() {
133        *p = it.next()?.parse().ok()?;
134    }
135    let pc_mode = it.next()?.trim() == "1";
136    // Campo da trava é opcional: firmware anterior não mandava.
137    let trava = it.next().map(|f| f.trim() == "1").unwrap_or(false);
138    Some(Status { pose, pc_mode, trava })
139}

141/// `B<versão>;boots=<n>;limb=<min>;<max>;limo=...;limc=...;limg=...`.
142///
143/// Os limites vêm em pares chave/valor (`limb=5;175`), uma chave por junta;
144/// chaves desconhecidas são ignoradas para permitir firmware mais novo.
145fn parse_diag(line: &str) -> Option<Diag> {
146    let parts: Vec<&str> = line[1..].split(';').collect();
147    let version = parts.first()?.to_string();
148    let mut boots = 0;
149    let mut limits = [[0.0, 180.0]; 4];
150    for i in 0..parts.len() {
151        let part = parts[i];
152        if let Some(v) = part.strip_prefix("boots=") {
153            boots = v.parse().ok()?;
154            continue;
155        }
156        let Some((key, min)) = part.split_once('=') else { continue };
157        let idx = match key {
158            "limb" => 0,
159            "limo" => 1,
160            "limc" => 2,
161            "limg" => 3,
162            _ => continue,
163        };
164        limits[idx] = [min.parse().ok()?, parts.get(i + 1)?.parse().ok()?];
165    }
166    Some(Diag { version, boots, limits })
167}

169#[cfg(test)]
170mod tests {
171    use super::*;
172
173    #[test]
174    fn status_do_firmware() {
175        let msg = parse_line("S95;63;154;10;1;0\n").unwrap();
176        assert_eq!(
177            msg,
178            Msg::Status(Status { pose: [95.0, 63.0, 154.0, 10.0], pc_mode: true, trava: false })
179        );
180        // firmware antigo, sem o campo da trava
181        let velho = parse_line("S95;63;154;10;0").unwrap();
182        assert_eq!(velho, Msg::Status(Status { pose: [95.0, 63.0, 154.0, 10.0], pc_mode: false, trava: false }));
183    }
184
185    #[test]
186    fn diag_do_firmware() {
187        let msg = parse_line("B1.3.0;boots=7;limb=5;175;limo=0;180;limc=0;180;limg=0;90").unwrap();
188        match msg {
189            Msg::Diag(d) => {
190                assert_eq!(d.version, "1.3.0");
191                assert_eq!(d.boots, 7);
192                assert_eq!(d.limits[0], [5.0, 175.0]);
193                assert_eq!(d.limits[3], [0.0, 90.0]);
194            }
195            other => panic!("esperava diag, veio {other:?}"),
196        }
197    }
198
199    #[test]
200    fn texto_livre_e_lixo() {
201        assert_eq!(
202            parse_line("firmware v1.3.0 boots=2"),
203            Some(Msg::Text("firmware v1.3.0 boots=2".into()))
204        );
205        assert_eq!(parse_line(""), None);
206        assert_eq!(parse_line("S95;63"), None);
207    }
208
209    #[test]
210    fn sticks_limitam() {
211        assert_eq!(cmd_sticks([120, -150, 0, 37]), "#V100;-100;0;37\n");
212    }
213
214    #[test]
215    fn alvos_arredondam() {
216        assert_eq!(cmd_targets([90.4, 59.6, 45.0, 20.5]), "#P90;60;45;21\n");
217    }
