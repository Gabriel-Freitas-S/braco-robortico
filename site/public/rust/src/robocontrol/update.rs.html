1//! Checagem das versões publicadas: app (GitHub Releases) e firmware (`main`).
2//!
3//! Roda em uma thread por checagem, entrega o resultado por `mpsc` e morre —
4//! o mesmo modelo de [`crate::link`]. A UI nunca espera rede: se não houver
5//! conexão, o resultado chega vazio e o app segue com o firmware embutido.
6//!
7//! # Por que não a API do GitHub
8//!
9//! A API sem token tem teto de 60 requisições por hora **por IP** e devolve 403
10//! quando estoura (comum em laboratório atrás de NAT). Aqui a versão do app vem
11//! do redirect de `github.com/.../releases/latest` — HTML, sem cota — e o
12//! firmware vem de `raw.githubusercontent.com`, também sem cota.
13//!
14//! # Segurança
15//!
16//! - Só HTTPS e só URLs fixas neste módulo; nenhuma resposta vira URL. A tag do
17//!   app passa por [`firmware::valida_versao`] antes de montar o link da
18//!   release com [`url_release`], e o destino final de cada redirect é
19//!   conferido contra o prefixo esperado.
20//! - Com timeouts e teto de leitura: host que não responde ou resposta gigante
21//!   não travam a UI nem estouram a memória.
22//! - Nada é executado: o firmware é texto validado por [`firmware::validar`] e
23//!   salvo como arquivo; uma versão nova do app só abre a página da release no
24//!   navegador. Não há auto-update.
25//! - Falha vira linha no log, sem popup — a checagem é um extra, não um
26//!   pré-requisito.
27//!
28//! # Limites conhecidos
29//!
30//! O firmware é lido de `main`. `shortcut: main é a versão publicada neste
31//! projeto; se deixar de ser, troque a URL por tags fw-v*`. Se o release mais
32//! recente do repositório for de firmware (tag fora de `app-v`), a versão do
33//! app fica desconhecida — de propósito, para não anunciar update falso.
34
35use crate::firmware;
36use std::io::Read;
37use std::sync::mpsc::{self, Receiver};
38use std::time::Duration;

40/// Repositório de onde saem binários e firmware.
41pub const REPO: &str = "Gabriel-Freitas-S/braco-robortico";
42
43const URL_ULTIMA_RELEASE: &str =
44    "https://github.com/Gabriel-Freitas-S/braco-robortico/releases/latest";
45const PREFIXO_TAG: &str = "https://github.com/Gabriel-Freitas-S/braco-robortico/releases/tag/";
46const URL_FIRMWARE: &str = "https://raw.githubusercontent.com/Gabriel-Freitas-S/braco-robortico/main/firmware/braco_4dof/braco_4dof.ino";
47const PREFIXO_RAW: &str = "https://raw.githubusercontent.com/Gabriel-Freitas-S/braco-robortico/";
48
49/// Teto de leitura do sketch (bytes). O arquivo real tem ~60 KB.
50const TETO_SKETCH: u64 = 128 * 1024;
51
52/// Identificação exigida pelos serviços do GitHub.
53const AGENTE: &str = concat!("robocontrol/", env!("CARGO_PKG_VERSION"));
54
55/// Versão do app em execução.
56pub fn versao_app() -> &'static str {
57    env!("CARGO_PKG_VERSION")
58}
59
60/// O que o repositório oferece hoje, já baixado.
61#[derive(Clone, Debug, Default)]
62pub struct Novidades {
63    /// Tag `app-vX.Y.Z` do release mais recente, se houver algum.
64    pub app: Option<String>,
65    /// `FW_VER` do sketch em `main`.
66    pub firmware: Option<String>,
67    /// Texto do sketch de `main`, para o botão "Salvar".
68    pub sketch: Option<String>,
69    /// Primeira falha de rede/parse; o que funcionou continua válido.
70    pub erro: Option<String>,
71}
72
73/// Dispara a checagem em segundo plano. Uma mensagem e a thread termina.
74pub fn checar() -> Receiver<Novidades> {
75    let (tx, rx) = mpsc::channel();
76    std::thread::spawn(move || {
77        let mut n = Novidades::default();
78
79        match ultima_app() {
80            Ok(Some(v)) => n.app = Some(v),
81            Ok(None) => {} // nenhum release publicado ainda: silêncio
82            Err(e) => n.erro = Some(e),
83        }
84
85        match baixa_sketch() {
86            Ok(corpo) => match firmware::validar(&corpo) {
87                Some(v) => {
88                    n.firmware = Some(v.to_string());
89                    n.sketch = Some(corpo);
90                }
91                None => n.erro = Some("sketch de main não parece válido".into()),
92            },
93            Err(e) => {
94                n.erro.get_or_insert(e);
95            }
96        }
97
98        let _ = tx.send(n);
99    });
100    rx
101}

103/// Link da página da release de uma versão já validada.
104pub fn url_release(versao: &str) -> Option<String> {
105    firmware::valida_versao(versao)?;
106    Some(format!(
107        "https://github.com/{REPO}/releases/tag/app-v{versao}"
108    ))
109}

111/// Versão embutida no destino de `releases/latest`; `None` se o release mais
112/// recente não for do app (ex.: um release de firmware).
113fn versao_app_da_url(url: &str) -> Option<&str> {
114    url.strip_prefix(PREFIXO_TAG)?
115        .strip_prefix("app-v")
116        .and_then(firmware::valida_versao)
117}
118
119/// Última versão do app publicada, seguindo o redirect de `releases/latest`.
120fn ultima_app() -> Result<Option<String>, String> {
121    let resposta = match agente()
122        .get(URL_ULTIMA_RELEASE)
123        .set("User-Agent", AGENTE)
124        .call()
125    {
126        Ok(r) => r,
127        //  Sem nenhum release o GitHub responde 404: é o estado inicial, não
128        //  um erro de rede, e não deve poluir o log.
129        Err(ureq::Error::Status(404, _)) => return Ok(None),
130        Err(e) => return Err(format!("releases: {e}")),
131    };
132    Ok(versao_app_da_url(resposta.get_url()).map(str::to_string))
133}
134
135/// Baixa o sketch de `main` (mesma fonte que o firmware embutido).
136fn baixa_sketch() -> Result<String, String> {
137    let resposta = agente()
138        .get(URL_FIRMWARE)
139        .set("User-Agent", AGENTE)
140        .call()
141        .map_err(|e| format!("firmware: {e}"))?;
142    let url = resposta.get_url();
143    if !url.starts_with(PREFIXO_RAW) {
144        return Err(format!("firmware: destino inesperado ({url})"));
145    }
146    let mut corpo = String::new();
147    resposta
148        .into_reader()
149        .take(TETO_SKETCH)
150        .read_to_string(&mut corpo)
151        .map_err(|e| format!("firmware: {e}"))?;
152    Ok(corpo)
153}
154
155fn agente() -> ureq::Agent {
156    ureq::AgentBuilder::new()
157        .timeout_connect(Duration::from_secs(5))
158        .timeout_read(Duration::from_secs(5))
159        .build()
160}

162#[cfg(test)]
163mod tests {
164    use super::*;
165
166    #[test]
167    fn versao_do_app_vem_so_de_tag_app_v() {
168        assert_eq!(
169            versao_app_da_url(
170                "https://github.com/Gabriel-Freitas-S/braco-robortico/releases/tag/app-v0.2.0"
171            ),
172            Some("0.2.0")
173        );
174        // Release de firmware não é versão do app.
175        assert_eq!(
176            versao_app_da_url(
177                "https://github.com/Gabriel-Freitas-S/braco-robortico/releases/tag/v1.4.0"
178            ),
179            None
180        );
181        // Host/prefixo diferente ou tag fora do formato: ignora.
182        assert_eq!(
183            versao_app_da_url("https://evil.example/releases/tag/app-v9.9.9"),
184            None
185        );
186        assert_eq!(
187            versao_app_da_url(
188                "https://github.com/Gabriel-Freitas-S/braco-robortico/releases/tag/app-v0.2"
189            ),
190            None
191        );
192    }
193
194    #[test]
195    fn url_da_release_recusa_tag_maliciosa() {
196        assert_eq!(
197            url_release("0.2.0").as_deref(),
198            Some("https://github.com/Gabriel-Freitas-S/braco-robortico/releases/tag/app-v0.2.0")
199        );
200        assert!(url_release("0.2.0/../../evil").is_none());
201        assert!(url_release("0.2.0?x=1").is_none());
202        assert!(url_release("").is_none());
203    }
204
205    /// Checagem real, mesma função do boot (TLS + URLs + parse). Fora do CI
206    /// porque depende de rede: `cargo test --release -- --ignored`.
207    #[test]
208    #[ignore = "usa a rede"]
209    fn checagem_real_baixa_o_sketch() {
210        let n = checar()
211            .recv_timeout(Duration::from_secs(20))
212            .expect("a thread não respondeu");
213        let fw = n.firmware.expect("versão do firmware não lida");
214        let sketch = n.sketch.expect("sketch não baixado");
215        assert_eq!(firmware::versao_de(&sketch), Some(fw.as_str()));
216        assert!(n.erro.is_none(), "erro inesperado: {:?}", n.erro);
217    }
